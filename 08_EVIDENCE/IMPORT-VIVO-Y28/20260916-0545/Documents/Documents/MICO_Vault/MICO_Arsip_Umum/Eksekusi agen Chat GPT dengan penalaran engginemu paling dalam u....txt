4. Generator laporan PEP

============================================================

cat > ~/JDEQ/PROJECT_CONTROL/PEP/generate_pep.py << 'PYEOF'
import json
import os
from datetime import datetime

def main():
# Path file input
workflow_path = os.path.expanduser("/JDEQ/PROJECT_CONTROL/WORKFLOW/workflow.json")
action_plan_path = os.path.expanduser("/JDEQ/PROJECT_CONTROL/PEP/action_plan.json")
output_path = os.path.expanduser("~/JDEQ/DATA_OUT/PEP_REPORT/PEP_REPORT.md")

# Maca data with open(workflow_path, 'r') as f: workflow_data = json.load(f) with open(action_plan_path, 'r') as f: action_plan_data = json.load(f) report = [] report.append("# PROJECT EXECUTION PLAN - JDEQ") report.append(f"**Tanggal:** {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}") report.append("") # Workflow report.append("## Workflow Proyek") for stage in workflow_data["stages"]: report.append(f"### {stage['id']} - {stage['name']}") report.append(f"- **Input:** {stage['input']}") report.append(f"- **Proses:** {stage['process']}") report.append(f"- **Output:** {stage['output']}") report.append(f"- **Status:** {stage['status']}") report.append("") # Action Plan plan = action_plan_data["project_execution_plan"] report.append("## Action Plan") report.append(f"**Tujuan:** {plan['objective']}") report.append("") for ms in plan["milestone"]: report.append(f"### {ms['id']} - {ms['name']}") report.append(f"- **PIC:** {ms['PIC']}") report.append(f"- **Target:** {ms['target']}") report.append("- **Tindakan:**") for act in ms["action"]: report.append(f" - {act}") report.append("") # Simpen os.makedirs(os.path.dirname(output_path), exist_ok=True) with open(output_path, 'w') as f: f.write('\n'.join(report)) print("[PEP] REPORT GENERATED") print(f"[PEP] {output_path}") 

if name == "main":
main()
