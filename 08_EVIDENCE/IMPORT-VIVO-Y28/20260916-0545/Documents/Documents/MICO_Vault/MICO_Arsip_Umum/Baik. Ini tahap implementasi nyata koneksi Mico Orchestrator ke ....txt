🧭 FASE 5 — MICO ROUTER CONNECTOR (CORE LINK)
 MICO EXECUTOR ENGINE
nano ~/JDEQ/kernel_engine/mico/dispatcher.py import subprocess import json AGENT_PATH = "~/JDEQ/agents" AGENT_FLOW = [ "01_intake_parser", "02_structural_engine", "03_civil_mapper", "06_cost_engine", "07_schedule_engine", "08_risk_engine", "12_output_generator" ] def run_agent(agent, payload): path = f"{AGENT_PATH}/{agent}/run.py" cmd = ["python", path, json.dumps(payload)] result = subprocess.run(cmd, capture_output=True, text=True) return result.stdout def execute_pipeline(task): data = task for agent in AGENT_FLOW: data = run_agent(agent, data) return data if __name__ == "__main__": task = {"project": "civil_to_digital_system"} print(execute_pipeline(task))
