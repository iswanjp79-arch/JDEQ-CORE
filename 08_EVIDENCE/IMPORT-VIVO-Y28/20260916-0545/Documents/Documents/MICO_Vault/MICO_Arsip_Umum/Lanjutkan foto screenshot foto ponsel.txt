# Modul Cathleen Brain JARVIS 685 dalam Ekosistem JEQ
import requests
import json
from jeq_heuristik import filer_data_madinah

def jarvis_collect_data(keyword):
    # Ambil data dari berbagai sumber
    sumber = ["https://api.lokal-indonesia.com", "https://api.ibm.com", "https://api.jeq-local.id"]
    semua_data = []
    for s in sumber:
        res = requests.get(f"{s}/search?q={keyword}")
        semua_data.extend(res.json()["data"])

    # Filter dengan Heuristik Madinah
    data_bersih = filer_data_madinah(semua_data)

    # Kirim ke Gatewanx untuk distribusi
    with open("$HOME/jeq/data_jarvis.json", "w") as f:
        json.dump(data_bersih, f)
    print("=== DATA JARVIS TERFILTRASI & DISIAPKAN ===")
    return data_bersih

# Panggil sesuai kebutuhan ekosistem
jarvis_collect_data("pengembangan pariwisata digital")
