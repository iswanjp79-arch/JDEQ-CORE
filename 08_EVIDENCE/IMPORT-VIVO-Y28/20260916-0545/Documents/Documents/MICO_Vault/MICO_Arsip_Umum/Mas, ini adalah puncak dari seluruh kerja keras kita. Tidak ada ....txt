```bash
cat >> ~/MICO-JDEQ-EOS/infrastructure/linux_app_manager.py << 'EOF'
# Tambahan untuk Docker, npm, Git
PACKAGE_MANAGERS.update({
    "docker": {"install": "docker pull {}", "remove": "docker rmi {}", "list": "docker images --format '{{.Repository}}:{{.Tag}}'"},
    "npm": {"install": "npm install -g {}", "remove": "npm uninstall -g {}", "list": "npm list -g --depth=0 2>/dev/null"},
    "git": {"install": "git clone {} {}", "remove": "rm -rf {}", "list": "ls -d */"}
})
def check_dependency(app_name):
    """Cek apakah aplikasi terinstal (untuk dependency resolver)."""
    result = subprocess.run(["which", app_name], capture_output=True, text=True)
    return result.returncode == 0
def resolve_dependencies(app_name, manager="apt"):
    """Resolver dependensi sederhana untuk apt."""
    if manager == "apt":
        deps = subprocess.run(["apt-cache", "depends", app_name], capture_output=True, text=True)
        return deps.stdout if deps.returncode == 0 else "Tidak dapat resolve dependensi."
    return "Resolver belum tersedia untuk manager ini."
EOF
```
