🧩 ARSITEKTUR DAEMON
[ TIMER LOOP ] ↓ [ ENV SCANNER ] ↓ [ PYTHON RESOLVER CHECK ] ↓ [ VENV DETECTOR ] ↓ [ PATH AUDITOR ] ↓ [ CONFLICT ENGINE ] ↓ [ DECISION LAYER ] ↓ [ ACTION HANDLER ] ↓ [ LOG WRITER ] 
 1. CORE RULES (ANTI DUPLICATE POLICY)
rules: rule_single_python_core: description: "Hanya satu Python 3.12 core diperbolehkan" check: - python_executable_path rule_single_venv: description: "Hanya satu venv aktif" check: - VIRTUAL_ENV variable - active env directory rule_path_consistency: description: "PATH tidak boleh berubah-ubah antar session" check: - PATH snapshot hash rule_no_pyenv_conflict: description: "pyenv tidak boleh override runtime" check: - pyenv active state rule_no_orphan_env: description: "env tanpa pyvenv.cfg dianggap orphan" check: - filesystem scan ~/JDEQ 
 2. DETECTION ENGINE (LOGIKA INTI)
import os import subprocess import hashlib JDEQ_HOME = os.path.expanduser("~/JDEQ") def get_python_path(): return subprocess.getoutput("which python") def get_virtual_env(): return os.environ.get("VIRTUAL_ENV", None) def hash_path(): return hashlib.sha256(os.environ.get("PATH","").encode()).hexdigest() def scan_venv(): candidates = [] for root, dirs, files in os.walk(JDEQ_HOME): if "pyvenv.cfg" in files: candidates.append(root) return candidates def detect_conflict(): issues = [] python_path = get_python_path() venv = get_virtual_env() venvs = scan_venv() if len(venvs) > 1: issues.append("MULTIPLE_VENV_DETECTED") if venv and len(venvs) == 0: issues.append("ORPHAN_VENV_STATE") if "pyenv" in python_path: issues.append("PYENV_RUNTIME_CONFLICT") return issues 
⚙️ 3. DECISION ENGINE
def decision_engine(issues): actions = [] if "MULTIPLE_VENV_DETECTED" in issues: actions.append("DELETE_EXTRA_VENV_KEEP_NEWEST") if "ORPHAN_VENV_STATE" in issues: actions.append("DEACTIVATE_VENV_FORCE_RESET") if "PYENV_RUNTIME_CONFLICT" in issues: actions.append("LOCK_PYENV") if not issues: actions.append("SYSTEM_OK") return actions 
🛠️ 4. ACTION HANDLER (SAFE MODE)
def execute_actions(actions): for action in actions: if action == "DELETE_EXTRA_VENV_KEEP_NEWEST": print("[SVD] Cleaning duplicate venv...") if action == "DEACTIVATE_VENV_FORCE_RESET": os.system("unset VIRTUAL_ENV") if action == "LOCK_PYENV": os.system("export PYENV_VERSION=system") if action == "SYSTEM_OK": print("[SVD] System stable") 
 5. DAEMON LOOP (TERMUX FRIENDLY)
import time def daemon_loop(): while True: issues = detect_conflict() actions = decision_engine(issues) execute_actions(actions) with open(os.path.expanduser("~/JDEQ/logs/svd.log"), "a") as f: f.write(f"{issues} -> {actions}\n") time.sleep(30) # ringan untuk Vivo Y28
