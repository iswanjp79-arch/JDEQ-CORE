2. Market Data Fetcher (market_data.py)

```python
import requests, time, json, os
from datetime import datetime

class MarketData:
    CACHE_FILE = os.path.expanduser("~/.jdeq_trading/data/price_cache.json")
    CACHE_DURATION = 60  # detik
    
    def __init__(self):
        self.cache = self.load_cache()
    
    def load_cache(self):
        try:
            with open(self.CACHE_FILE) as f:
                return json.load(f)
        except:
            return {}
    
    def save_cache(self):
        with open(self.CACHE_FILE, 'w') as f:
            json.dump(self.cache, f)
    
    def get_price(self, coin_id, vs_currency="idr"):
        now = time.time()
        key = f"{coin_id}_{vs_currency}"
        if key in self.cache and now - self.cache[key]['timestamp'] < self.CACHE_DURATION:
            return self.cache[key]['price']
        # Fetch from API
        try:
            url = f"https://api.coingecko.com/api/v3/simple/price?ids={coin_id}&vs_currencies={vs_currency}"
            r = requests.get(url, timeout=10)
            data = r.json()
            price = data[coin_id][vs_currency]
            self.cache[key] = {'price': price, 'timestamp': now}
            self.save_cache()
            return price
        except Exception as e:
            print(f"Error fetching {coin_id}: {e}")
            return None
    
    def get_historical(self, coin_id, days=30):
        # Ambil data historis kanggo indikator (sederhana)
        url = f"https://api.coingecko.com/api/v3/coins/{coin_id}/market_chart?vs_currency=idr&days={days}"
        try:
            r = requests.get(url, timeout=15)
            return r.json()['prices']  # list of [timestamp, price]
        except:
            return None
```

3. Analyst Agent (LLM-based) (analyst_agent.py)

```python
import json, requests, subprocess

class AnalystAgent:
    def __init__(self, model="phi3:mini"):
        self.model = model
        self.ollama_url = "http://localhost:11434/api/generate"
    
    def ask_llama(self, prompt):
        payload = {
            "model": self.model,
            "prompt": prompt,
            "stream": False,
            "options": {"temperature":0.3, "max_tokens":150}
        }
        try:
            r = requests.post(self.ollama_url, json=payload, timeout=30)
            return r.json()["response"]
        except:
            return None
    
    def analyze(self, coin, price, rsi, sma_short, sma_long):
        prompt = f"""Anda adalah analis pasar crypto profesional. Berdasarkan data berikut untuk {coin}:
- Harga saat ini: {price}
- RSI (14 hari): {rsi:.2f}
- SMA 50: {sma_short:.2f}
- SMA 200: {sma_long:.2f}

Berikan rekomendasi: BELI, JUAL, atau TAHAN. Sertakan alasan singkat dan confidence level (0-100%). Output JSON: {{"action":"BUY/SELL/HOLD", "confidence":0-100, "reason":"..."}}"""
        response = self.ask_llama(prompt)
        if response:
            try:
                # Ambil JSON dari respons
                import re
                match = re.search(r'\{.*?\}', response, re.DOTALL)
                if match:
                    return json.loads(match.group())
            except:
                pass
        return {"action":"HOLD", "confidence":0, "reason":"Analisis gagal"}
```

4. Risk Agent (risk_agent.py)

```python
class RiskAgent:
    def __init__(self, max_loss_percent=5, max_position_size=0.3):
        self.max_loss_percent = max_loss_percent
        self.max_position_size = max_position_size
    
    def evaluate(self, portfolio, price, action):
        # portfolio: dict with cash, holdings
        if action == "BUY":
            # Hitung ukuran posisi maksimal
            affordable = portfolio['cash'] * self.max_position_size
            return min(affordable, portfolio['cash'] * 0.5)  # misal
        elif action == "SELL":
            # Berapa banyak yang bisa dijual
            return portfolio['holdings']
        else:
            return 0
    
    def check_stop_loss(self, portfolio, entry_price, current_price):
        if entry_price == 0:
            return False
        loss_percent = (entry_price - current_price) / entry_price * 100
        return loss_percent > self.max_loss_percent
```

5. Executor Agent (executor_agent.py) – Paper Trading Mode

```python
import time, json, os

class PaperExecutor:
    def __init__(self, initial_cash=1000000):
        self.portfolio = {
            'cash': initial_cash,
            'holdings': 0,
            'history': []
        }
        self.file = os.path.expanduser("~/.jdeq_trading/data/portfolio.json")
        self.load()
    
    def load(self):
        try:
            with open(self.file) as f:
                self.portfolio = json.load(f)
        except:
            pass
    
    def save(self):
        with open(self.file, 'w') as f:
            json.dump(self.portfolio, f)
    
    def execute(self, action, price, amount):
        timestamp = time.time()
        if action == "BUY":
            cost = amount * price
            if cost <= self.portfolio['cash']:
                self.portfolio['cash'] -= cost
                self.portfolio['holdings'] += amount
                self.portfolio['history'].append({'type':'BUY','price':price,'amount':amount,'time':timestamp})
                self.save()
                return True
        elif action == "SELL":
            if amount <= self.portfolio['holdings']:
                revenue = amount * price
                self.portfolio['cash'] += revenue
                self.portfolio['holdings'] -= amount
                self.portfolio['history'].append({'type':'SELL','price':price,'amount':amount,'time':timestamp})
                self.save()
                return True
        return False
    
    def get_value(self, current_price):
        return self.portfolio['cash'] + self.portfolio['holdings'] * current_price
```
