import asyncio
import aiohttp
import time
from typing import List, Optional
# Metafora: "Corong Pipa Dinamis" - bisa mampat atau mengembang sesuai kebutuhan
class CorongDinamikSemaphore:
    def __init__(self, kapasitas_awal: int, kapasitas_maks: int, threshold_beban: float = 0.8):
        self.kapasitas_awal = kapasitas_awal  # Pipa dalam keadaan mampat
        self.kapasitas_maks = kapasitas_maks  # Pipa mengembang ke ukuran maks
        self.threshold_beban = threshold_beban  # Batasan untuk mulai mengembang
        self.semaphore = asyncio.Semaphore(kapasitas_awal)
        self.jumlah_tugas_aktif = 0
        self.lock = asyncio.Lock()

    # Logika: Jika beban mendekati batas, corong mengembang; jika beban rendah, mampat kembali
    async def sesuaikan_kapasitas(self):
        async with self.lock:
            beban_sekarang = self.jumlah_tugas_aktif / self.semaphore._value
            if beban_sekarang >= self.threshold_beban and self.semaphore._value < self.kapasitas_maks:
                # Corong mengembang - tambah kapasitas
                tambah_kapasitas = min(2, self.kapasitas_maks - self.semaphore._value)
                self.semaphore._value += tambah_kapasitas
                print(f"[Corong Mengembang] Kapasitas sekarang: {self.semaphore._value}")
            elif beban_sekarang < self.threshold_beban / 2 and self.semaphore._value > self.kapasitas_awal:
                # Corong Mampat - kurangi kapasitas ke ukuran awal
                kurangi_kapasitas = min(2, self.semaphore._value - self.kapasitas_awal)
                self.semaphore._value -= kurangi_kapasitas
                print(f"[Corong Mampat] Kapasitas sekarang: {self.semaphore._value}")

    async def __aenter__(self):
        async with self.lock:
            self.jumlah_tugas_aktif += 1
        await self.sesuaikan_kapasitas()
        await self.semaphore.acquire()
        return self

    async def __aexit__(self, exc_type, exc, tb):
        self.semaphore.release()
        async with self.lock:
            self.jumlah_tugas_aktif -= 1
        await self.sesuaikan_kapasitas()


async def fetch_data(session: aiohttp.ClientSession, url: str, corong_sem: CorongDinamikSemaphore) -> str:
    """Fungsi ambil data dengan mengikuti aliran corong pipa dinamis"""
    async with corong_sem:
        try:
            start_waktu = time.time()
            async with session.get(url, timeout=10) as response:
                data = await response.text()
                durasi = round(time.time() - start_waktu, 2)
                return f"[Sukses] URL: {url[:30]}... | Durasi: {durasi} detik"
        except Exception as e:
            return f"[Error] URL: {url[:30]}... | Pesan: {str(e)}"


async def main(jumlah_tugas: int = 50, kapasitas_awal: int = 5, kapasitas_maks: int = 20):
    # Inisialisasi corong - pipa awal mampat, bisa mengembang hingga batas maks
    corong_pipa = CorongDinamikSemaphore(
        kapasitas_awal=kapasitas_awal,
        kapasitas_maks=kapasitas_maks,
        threshold_beban=0.7
    )

    # Buat daftar URL (seperti aliran data yang masuk ke corong)
    urls = [f"https://httpbin.org/get?param={i}" for i in range(jumlah_tugas)]
    print(f"[Mulai] Total tugas: {jumlah_tugas} | Corong siap dengan kapasitas awal {kapasitas_awal}")

    async with aiohttp.ClientSession() as session:
        # Buat tugas-tugas seperti aliran yang melalui corong
        tasks: List[asyncio.Task] = [
            asyncio.create_task(fetch_data(session, url, corong_pipa))
            for url in urls
        ]

        # Kumpulkan hasil seperti data yang keluar dari ujung corong
        results = await asyncio.gather(*tasks)

    # Tampilkan ringkasan
    print("\n[Ringkasan Hasil]")
    sukses = sum(1 for res in results if res.startswith("[Sukses]"))
    error = len(results) - sukses
    print(f"Sukses: {sukses} | Error: {error}")


if __name__ == "__main__":
    # Jalankan mesin corong
    asyncio.run(main(jumlah_tugas=50))
