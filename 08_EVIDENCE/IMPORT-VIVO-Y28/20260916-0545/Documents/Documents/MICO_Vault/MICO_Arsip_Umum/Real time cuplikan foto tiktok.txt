def klasifikasi_file_quantum(path_file):

    prediksi_klasik = model_klasik.predict(fitur_file)

    # Validasi dengan Quantum

    qc.h([0,1])
    qc.rx(prediksi_klasik[0][0] * np.pi, 0)
    qc.rx(prediksi_klasik[0][1] * np.pi, 1)
    qc.measure_all()
