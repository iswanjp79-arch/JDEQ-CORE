## **PART 2: GHOST PROTOCOL VIA GOOGLE APP SCRIPT**

```javascript
//  GAS_GHOST_PROTOCOL.js
// Simpan di Google Apps Script

class WitnessGrid {
  constructor() {
    this.layers = 3;
    this.shardSize = 1024; // bytes per shard
  }
  
  shredData(data) {
    // 1. Enkripsi dengan 3 lapis
    const encrypted = Utilities.base64Encode(
      Utilities.computeHmacSha256Signature(
        Utilities.base64Encode(
          Utilities.computeHmacSha256Signature(data, "LAYER1")
        ), "LAYER2"
      )
    );
    
    // 2. Pecah menjadi shards
    const shards = [];
    for (let i = 0; i < encrypted.length; i += this.shardSize) {
      shards.push(encrypted.substring(i, i + this.shardSize));
    }
    
    // 3. Tambahkan metadata witness
    const timestamp = new Date().getTime();
    const witnessHash = Utilities.computeDigest(
      Utilities.DigestAlgorithm.SHA_256, 
      timestamp.toString()
    ).toString();
    
    return {
      shards: shards,
      totalShards: shards.length,
      witnessHash: witnessHash,
      timestamp: timestamp
    };
  }
  
  hideInGmail(shards) {
    // Simpan di Draft/Trash sebagai "sampah"
    const emails = [];
    
    shards.forEach((shard, index) => {
      // Buat draft email
      const draft = GmailApp.createDraft(
        `ghost${index}@witness.grid`,
        `SPAM-${Math.random().toString(36).substring(2)}`,
        shard,
        {
          htmlBody: `<!-- WITNESS_GRID_SHARD_${index} -->`
        }
      );
      
      emails.push(draft.getId());
    });
    
    return emails;
  }
  
  recoverFromGmail() {
    // Recover dari "sampah"
    const drafts = GmailApp.getDrafts();
    const shards = [];
    
    drafts.forEach(draft => {
      const body = draft.getMessage().getBody();
      if (body.includes('WITNESS_GRID_SHARD')) {
        shards.push(body);
      }
    });
    
    return this.reassembleData(shards);
  }
}
```

```python
import time, psutil, json
from datetime import datetime

class BioTechnoMonitor:
    """Monitor kondisi biologis melalui sensor HP"""
    
    def __init__(self):
        self.stress_level = 0
        self.sleep_hours = 0
        self.last_break = time.time()
        
    def detect_stress(self):
        """Deteksi stres dari penggunaan HP"""
        metrics = {
            "screen_time": self.get_screen_time(),
            "typing_speed": self.get_typing_pattern(),
            "battery_cycles": psutil.sensors_battery().cycle_count if hasattr(psutil.sensors_battery(), 'cycle_count') else 0,
            "cpu_temp": self.estimate_temperature()
        }
        
        # Heuristic stres detection
        stress_score = 0
        if metrics["screen_time"] > 6 * 3600:  # >6 jam
            stress_score += 40
        if time.time() - self.last_break > 2 * 3600:  # >2 jam tanpa break
            stress_score += 30
            
        self.stress_level = stress_score
        return stress_score
    
    def padi_pasungguh_logic(self, requested_action: str) -> dict:
        """AI menolak perintah berbahaya"""
        dangerous_actions = [
            "hack", "crack", "ddos", "exploit",
            "overclock", "delete all", "format",
            "bypass security", "illegal"
        ]
        
        current_stress = self.detect_stress()
        
        # Logika Padi: tumbuh hanya di kondisi tepat
        if current_stress > 60:
            return {
                "allowed": False,
                "reason": "🧘 Stres level tinggi. Istirahat dulu 15 menit.",
                "suggestion": "Coba: meditasi 5 menit atau minum air"

                "allowed": False,

                "allowed": False,
                "reason": " Kurang tidur mengurangi kualitas keputusan.",
                "suggestion": "Tidur 7-8 jam, lanjutkan besok"

        return {"allowed": True, "reason": " Kondisi optimal untuk kreativitas"}
```
