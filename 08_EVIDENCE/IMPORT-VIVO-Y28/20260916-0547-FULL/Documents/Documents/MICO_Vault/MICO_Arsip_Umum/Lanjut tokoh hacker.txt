import json

# Simulasi database ctOS dengan data NPC
def get_npc_data(npc_id):
    conn = sqlite3.connect('ctos_chicago.db')
    cursor = conn.cursor()
    cursor.execute("SELECT * FROM civilians WHERE id=?", (npc_id,))
    data = cursor.fetchone()
    conn.close()
    if data:
        return {
            "nama": data[1],
            "pendapatan": data[2],
            "riwayat_kriminal": json.loads(data[3]),
            "rekening_bank": data[4]
        }
    return None

import time
from pymodbus.client import ModbusTcpClient

# Simulasi kontrol lampu lalu lintas via Modbus (mirip ctOS)
def control_traffic_light(light_id, status):
    client = ModbusTcpClient('192.168.1.100')  # Alamat simulasi perangkat
    client.connect()
    # Status: 0 = merah, 1 = kuning, 2 = hijau
    client.write_register(address=light_id, value=status)
    print(f"Lampu lalu lintas {light_id} diatur ke status {status}")
    client.close()

# Contoh penggunaan

from scapy.all import sniff, IP, TCP

# Simulasi penyadapan pesan teks melalui jaringan
def sniff_messages(packet):
    if packet.haslayer(TCP) and packet.dport == 80:
        payload = packet[TCP].payload.load.decode('utf-8', errors='ignore')
        if "message=" in payload:
            print("Pesan terdeteksi:", payload.split("message=")[1].split("&")[0])

# Mulai menyadap jaringan lokal

import dronekit_sitl
from dronekit import connect, VehicleMode, LocationGlobalRelative

# Simulasi kontrol drone (mirip drone Marcus)
def control_drone():
    sitl = dronekit_sitl.start_default()
    connection_string = sitl.connection_string()
    vehicle = connect(connection_string, wait_ready=True)
    
    # Mengangkat drone
    vehicle.mode = VehicleMode("GUIDED")
    vehicle.armed = True
    vehicle.simple_takeoff(10)  # Naik ke ketinggian 10m
    time.sleep(5)
    
    # Bergerak ke lokasi target
    target = LocationGlobalRelative(-35.363261, 149.165230, 10)
    vehicle.simple_goto(target)
    print("Drone sedang mengarah ke target...")
    time.sleep(10)
    
    # Mendaratkan drone
    vehicle.mode = VehicleMode("LAND")
    vehicle.close()
    sitl.stop()

# Contoh penggunaan
control_drone()

import networkx as nx
import matplotlib.pyplot as plt

# Simulasi peta jaringan seperti "Hacker Vision" Marcus
def visualize_network():
    G = nx.Graph()
    # Menambahkan node: perangkat dan NPC
    nodes = ["Router Utama", "Lampu Lalu Lintas", "Kamera Keamanan", "NPC1", "NPC2"]
    G.add_nodes_from(nodes)
    # Menambahkan koneksi antar perangkat
    edges = [("Router Utama", "Lampu Lalu Lintas"), ("Router Utama", "Kamera Keamanan"),
             ("Kamera Keamanan", "NPC1"), ("Router Utama", "NPC2")]
    G.add_edges_from(edges)
    
    # Menampilkan peta jaringan
    plt.figure(figsize=(8, 6))
    nx.draw(G, with_labels=True, node_color="lightblue", font_size=10)
    plt.title("Peta Jaringan ctOS 2.0 (Bay Area)")
    plt.show()

# Contoh penggunaan
visualize_network()

1. MEMBUAT ALAT HACKING DARURAT DARI PERANGKAT SEHARI-HARI
 
- Konsep: Seperti membuat bom dari bahan dapur di serial lama, sekarang dia akan mengubah perangkat bekas (seperti ponsel lama, router bekas, atau sensor murah) menjadi alat hacking menggunakan Python.
- Simulasi dengan Python:
- Menggunakan library  pyserial  untuk mengontrol sensor dari perangkat bekas dan mengubahnya menjadi alat deteksi jaringan.
- Menggunakan  scapy  untuk membuat "alat pemindai jaringan portabel" dari ponsel yang sudah di-root, dengan AI yang dilatih untuk mendeteksi celah keamanan secara otomatis.
 
python  
from scapy.all import ARP, Ether, srp
import pandas as pd

# Alat deteksi perangkat di jaringan (dari ponsel lama)
def scan_network(ip_range):
    arp = ARP(pdst=ip_range)
    ether = Ether(dst="ff:ff:ff:ff:ff:ff")
    packet = ether/arp
    result = srp(packet, timeout=3, verbose=0)[0]

    devices = []
    for sent, received in result:
        devices.append({'IP': received.psrc, 'MAC': received.hwsrc})

    # AI untuk menganalisis apakah ada perangkat mencurigakan
    df = pd.DataFrame(devices)
    suspicious = df[df['MAC'].str.startswith('00:1A:2B')]  # Contoh pola mencurigakan
    if not suspicious.empty:
        print("Perangkat mencurigakan terdeteksi:", suspicious)
    return df

# Contoh penggunaan: Scan jaringan lokal
scan_network("192.168.1.1/24")
