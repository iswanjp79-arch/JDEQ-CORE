}

# Konversi data ke representasi kuantum
def personality_to_qubits(data):
    # Setiap kategori menjadi satu qubit
    qc = QuantumCircuit(4, 4)  # Qubit 0:Hobi, 1:Nilai, 2:Kuesioner, 3:Keterampilan
    
    # Encode data ke dalam keadaan kuantum
    hobi_score = len(data["hobi"]) / 5  # Normalisasi
    nilai_score = sum([1 for n in data["nilai_pribadi"] if n in ["Kerja Keras", "Jujur", "Inovatif"]]) / 3
    kuesioner_score = sum([1 for q in data["kuesioner_organisasi"] if q["jawaban"] == "Ya"]) / 4
    keterampilan_score = len(data["keterampilan_khusus"]) / 5
    
    # Terapkan gerakan kuantum berdasarkan skor
    qc.rx(hobi_score * np.pi, 0)
    qc.rx(nilai_score * np.pi, 1)
    qc.rx(kuesioner_score * np.pi, 2)
    qc.rx(keterampilan_score * np.pi, 3)
    
    # Entangle semua qubit untuk melihat hubungan antar karakteristik
    qc.cx(0, 1)
    qc.cx(1, 2)
    qc.cx(2, 3)
    
    return qc

# Jalankan simulasi kuantum
qc = personality_to_qubits(personality_data)
simulator = Aer.get_backend("statevector_simulator")
result = execute(qc, simulator).result()
statevector = result.get_statevector()

# Fungsi untuk pembentukan tim dengan Swarm Intelligence
class SwarmTeamFormation:
    def __init__(self, main_employee, others):
        self.main = main_employee
        self.others = others
        self.swarm_size = len(others) + 1
        self.positions = np.random.rand(self.swarm_size, 2)  # Posisi setiap anggota dalam ruang karakteristik
    
    def fitness(self, position):
        # Hitung kecocokan berdasarkan keahlian dan nilai
        keahlian_overlap = len(set(self.main["keterampilan_khusus"]) & set(self.others[int(position[0])]["keahlian"]))
        nilai_overlap = len(set(self.main["nilai_pribadi"]) & set(self.others[int(position[0])]["nilai"]))
        return (keahlian_overlap + nilai_overlap) / 6  # Normalisasi
    
    def update_positions(self, max_iter=50):
        for _ in range(max_iter):
            for i in range(self.swarm_size-1):
                current_fitness = self.fitness(self.positions[i])
                neighbor_fitness = self.fitness(self.positions[(i+1)%(self.swarm_size-1)])
                if neighbor_fitness > current_fitness:
                    self.positions[i] = self.positions[(i+1)%(self.swarm_size-1)] + np.random.randn(2)*0.1
        return self.positions
    
    def form_best_team(self):
        positions = self.update_positions()
        fitness_scores = [self.fitness(p) for p in positions[:-1]]
        best_members_idx = np.argsort(fitness_scores)[-2:]  # Pilih 2 anggota terbaik
        best_team = [self.others[i] for i in best_members_idx]
        best_team.insert(0, self.main)
        return best_team

# Jalankan pembentukan tim
swarm = SwarmTeamFormation(personality_data, other_employees)
best_team = swarm.form_best_team()

print("\n=== TIM KERJA OPTIMAL DENGAN SWARM INTELLIGENCE ===")
print("Anggota Tim:")
