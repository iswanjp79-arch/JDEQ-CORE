# --- Konfigurasi ---

MQTT_PORT = 1883
TOPIC = "allspark/awareness/events"
# Asumsi Interval Monitoring
MONITOR_INTERVAL = 10
# --- Logging Setup ---
logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')
# --- Database & MQTT Functions ---
def ensure_db():
"Memastikan database SQLite ada dan tabel events/state sudah dibuat."
conn = sqlite3.connect(DB)
c = conn.cursor()
# Tabel events untuk mencatat semua tindakan/status
c.execute('''
CREATE TABLE IF NOT EXISTS events (
id INTEGER PRIMARY KEY,
timestamp TEXT,
event_type TEXT,
data TEXT)'')
# Tabel state untuk menyimpan data key-value
c.execute('''
CREATE TABLE IF NOT EXISTS state (
key TEXT PRIMARY KEY,
value TEXT)'')conn.commit()return conn
def log_event(conn, event_type, data):
"Mencatat event ke database lokal."
timestamp = datetime.now().isoformat()
conn.execute("INSERT INTO events VALUES (NULL, ?, ?, ?)",
(timestamp, event_type, json.dumps(data)))
conn.commit()
logging.info(f"LOGGED: {event_type} - {data}")
def publish(client, event_type, data):
"Menerbitkan event ke MQTT broker."
payload = {
timestamp: datetime.now().isoformat(),
event_type: event_type,
data: data}
client.publish(TOPIC, json.dumps(payload), qos=1)
logging.info(f"PUBLISHED: {event_type}")
# --- Core Logic ---
def check_system():
"Mengambil metrik CPU, Memori, dan Disk."
try:metrics = {
cpu: psutil.cpu_percent(interval=1), # CPU usage over 1 second
mem: psutil.virtual_memory().percent,
disk: psutil.disk_usage('/').percent
}return metrics
except Exception as e:
logging.error(f"Error checking system metrics: {e}")
return {"cpu": 0, "mem": 0, "disk": 0}
def check_broker():
"Memverifikasi status layanan Mosquitto menggunakan systemctl."
try:
# Asumsi Termux menggunakan systemctl emulation (termux-services)
subprocess.check_call(["sh", "-c", "systemctl status mosquitto > /dev/null 2>&1"])
return True
except subprocess.CalledProcessError:
logging.warning("Mosquitto service is not running.")
return False
except FileNotFoundError:
logging.warning("systemctl command not found (Termux-services not active?)")
return False

--arg timestamp "$TIMESTAMP" \
--arg path "$path" \

source: "ManifestWatcher"
}')
