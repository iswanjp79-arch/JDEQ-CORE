from google import genai
import os

def muat_kunci():
    # Ambil kunci dari file brankas
    path = os.path.expanduser("~/.daftar_kunci.txt")
    if os.path.exists(path):
        with open(path, "r") as f:
            # Ambil hanya baris yang ada kunci AIza
            return [line.strip().split(". ")[1] for line in f 
                    if line.strip() and "AIza" in line]
    return []

def core_activation():
    kunci_list = muat_kunci()
    print("\n" + "="*35)
    print(" J.A.R.V.I.S MULTI-KEY VAULT ")
    print(f" Ditemukan {len(kunci_list)} Kunci Aktif ")
    print("="*35)

    for i, key in enumerate(kunci_list):
        try:
            client = genai.Client(api_key=key)
            response = client.models.generate_content(
                model="gemini-2.0-flash",
                contents="System check. Are you active?"
            )
            print(f"\n[✅ SUCCESS]: Kunci #{i+1} Berhasil!")
            print(f"[ REPLY] : {response.text}")
            return # Berhenti jika menemukan kunci yang bisa dipakai
        except Exception as e:
            if "429" in str(e):
                print(f"[⚠️ LIMIT]: Kunci #{i+1} Kuota Habis")
            else:
                print(f"[❌ ERROR]: Kunci #{i+1} Bermasalah")

    print("\n[❗ CRITICAL]: Semua kunci sudah habis kuota!")

if __name__ == "__main__":
    core_activation()
