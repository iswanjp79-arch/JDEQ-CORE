from datetime import datetime
# Load config
with open("config_blueprint19.json") as f:
config = json.load(f)
LOG_DIR = config["paths"]["log_dir"]
AUDIT_OUTPUT = config["paths"]["audit_output"]
ERROR_LIMIT = config["thresholds"]["error_frequency_limit"]
def load_logs():logs = []
for fname in os.listdir(LOG_DIR):
if fname.endswith(".json"):
with open(os.path.join(LOG_DIR, fname)) as f:
logs.append(json.load(f))
return logs
def introspect_errors(logs):
error_count = sum(1 for log in logs if log.get("type") == "error")
return {
total_logs: len(logs),
error_count: error_count,
status: "critical" if error_count > ERROR_LIMIT else "stable"
}
def generate_audit_report():
logs = load_logs()
introspection = introspect_errors(logs)
report = {
timestamp: datetime.now().isoformat(),
introspection: introspection,
notes: "Audit berdasarkan Blueprint Final 19: Kesadaran AI"
}
os.makedirs(os.path.dirname(AUDIT_OUTPUT), exist_ok=True)
with open(AUDIT_OUTPUT, "w") as f:
json.dump(report, f, indent=2)
print("Audit report generated:", AUDIT_OUTPUT)
if __name__ == "__main__":
generate_audit_report()
