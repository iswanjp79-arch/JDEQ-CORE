#!/data/data/com.termux/files/usr/bin/python3
import os, time, json, psutil, subprocess
from pathlib import Path

class AutopoieticWatchdog:
    def __init__(self, doctrine_path="doctrine.json"):
        with open(doctrine_path) as f:
            self.doctrine = json.load(f)
        self.thresholds = self.doctrine["thresholds"]
        self.actions = self.doctrine["actions"]
        self.processes = {}
    
    def check_health(self):
        # Monitor CPU, RAM, suhu
        cpu = psutil.cpu_percent()
        mem = psutil.virtual_memory().percent
        temp = self.get_temperature()
        if temp > self.thresholds["temp_max"]:
            self.execute_action("thermal_throttle")
        if mem > self.thresholds["ram_max"]:
            self.execute_action("kill_low_priority")
        if cpu > self.thresholds["cpu_max"]:
            self.execute_action("reduce_batch")
        return {"cpu": cpu, "mem": mem, "temp": temp}
    
    def get_temperature(self):
        try:
            out = subprocess.check_output("termux-battery-status", shell=True)
            return json.loads(out)["temperature"]
        except:
            return 30.0
    
    def execute_action(self, action):
        cmd = self.actions.get(action)
        if cmd:
            subprocess.Popen(cmd, shell=True)
    
    def run(self):
        while True:
            health = self.check_health()
            print(f"HEALTH: {health}")
            time.sleep(10)

if __name__ == "__main__":
    wd = AutopoieticWatchdog()
    wd.run()

import json, time, subprocess, threading
import paho.mqtt.client as mqtt

class SensorBridge:
    def __init__(self):
        self.priority = 1.0  # 1.0 = normal
        self.client = mqtt.Client()
        self.client.connect("localhost", 1883)
    
    def read_battery(self):
        try:
            data = json.loads(subprocess.check_output("termux-battery-status", shell=True))
            return data["percentage"], data["temperature"]
        except:
            return 50, 30
    
    def read_crypto(self):
        # contoh: ambil harga Bitcoin dari API (gunakan cache jika offline)
        try:
            import requests
            r = requests.get("https://api.coingecko.com/api/v3/simple/price?ids=bitcoin&vs_currencies=usd", timeout=2)
            return r.json()["bitcoin"]["usd"]
        except:
            return None
    
    def allocate_priority(self, temp):
        if temp > 38:
            self.priority = 0.3
        elif temp > 35:
            self.priority = 0.7
        else:
            self.priority = 1.0
        self.client.publish("jdeq/priority", str(self.priority))
    
    def run(self):
        while True:
            batt, temp = self.read_battery()
            crypto = self.read_crypto()
            self.allocate_priority(temp)
            payload = {
                "battery": batt,
                "temperature": temp,
                "crypto": crypto,
                "priority": self.priority,
                "timestamp": time.time()
            }
            self.client.publish("jdeq/sensors", json.dumps(payload))
            time.sleep(30)

if __name__ == "__main__":
    sb = SensorBridge()
    sb.run()

import sqlite3, json, networkx as nx
from cryptography.fernet import Fernet

class IdentityGraph:
    def __init__(self, db_path="identity.db", key_path="secret.key"):
        self.conn = sqlite3.connect(db_path)
        self.create_tables()
        self.graph = nx.Graph()
        self.load_graph()
        self.cipher = self.load_key(key_path)
    
    def create_tables(self):
        self.conn.execute("""
            CREATE TABLE IF NOT EXISTS nodes (
                id TEXT PRIMARY KEY,
                label TEXT,
                metadata TEXT,
                created REAL
            )
        """)
        self.conn.execute("""
            CREATE TABLE IF NOT EXISTS edges (
                source TEXT,
                target TEXT,
                relation TEXT,
                weight REAL,
                PRIMARY KEY (source, target)
            )
        """)
    
    def load_key(self, key_path):
        try:
            with open(key_path, "rb") as f:
                key = f.read()
        except:
            key = Fernet.generate_key()
            with open(key_path, "wb") as f:
                f.write(key)
        return Fernet(key)
    
    def add_memory(self, node_id, label, metadata):
        encrypted = self.cipher.encrypt(json.dumps(metadata).encode())
        self.conn.execute("INSERT OR REPLACE INTO nodes VALUES (?,?,?,?)",
                          (node_id, label, encrypted.decode(), time.time()))
        self.graph.add_node(node_id, label=label, metadata=metadata)
        self.conn.commit()
    
    def relate(self, source, target, relation, weight=1.0):
        self.conn.execute("INSERT OR REPLACE INTO edges VALUES (?,?,?,?)",
                          (source, target, relation, weight))
        self.graph.add_edge(source, target, relation=relation, weight=weight)
        self.conn.commit()
    
    def load_graph(self):
        # Load nodes
        cursor = self.conn.execute("SELECT id, label, metadata FROM nodes")
        for row in cursor:
            meta = json.loads(self.cipher.decrypt(row[2].encode()).decode())
            self.graph.add_node(row[0], label=row[1], metadata=meta)
        # Load edges
        cursor = self.conn.execute("SELECT source, target, relation, weight FROM edges")
        for row in cursor:
            self.graph.add_edge(row[0], row[1], relation=row[2], weight=row[3])
    
    def reflect(self):
        # Analisis graf: cari node paling terhubung, cluster, dll
        return {
            "num_nodes": self.graph.number_of_nodes(),
            "num_edges": self.graph.number_of_edges(),
            "central_nodes": sorted(nx.degree_centrality(self.graph).items(), key=lambda x: x[1], reverse=True)[:3]
        }

5. Self-Modifying Objective Compiler (objective_compiler.py)

```python
import random, json, copy

class ObjectiveDNA:
    def __init__(self, dna_string):
        self.genes = dna_string  # misal: "PROFIT>0.7;EXPLORE<0.3;STABILITY>0.5"
    
    def mutate(self, rate=0.1):
        # Mutasi sederhana: ubah angka atau operator
        genes = self.genes.split(';')
        new_genes = []
        for g in genes:
            if random.random() < rate:
                # Ganti angka
                parts = g.split(('>','<')[0])
                if '>' in g:
                    name, val = g.split('>')
                    new_val = max(0.1, min(1.0, float(val) + random.uniform(-0.1, 0.1)))
                    new_genes.append(f"{name}>{new_val:.2f}")
                elif '<' in g:
                    name, val = g.split('<')
                    new_val = max(0.1, min(1.0, float(val) + random.uniform(-0.1, 0.1)))
                    new_genes.append(f"{name}<{new_val:.2f}")
                else:
                    new_genes.append(g)
            else:
                new_genes.append(g)
        return ObjectiveDNA(';'.join(new_genes))
    
    def fitness(self, feedback):
        # feedback: dict of metrics, e.g., {"PROFIT":0.8, "EXPLORE":0.2, "STABILITY":0.9}
        score = 0.0
        for g in self.genes.split(';'):
            if '>' in g:
                name, target = g.split('>')
                if name in feedback and feedback[name] >= float(target):
                    score += 1
            elif '<' in g:
                name, target = g.split('<')
                if name in feedback and feedback[name] <= float(target):
                    score += 1
        return score / len(self.genes.split(';'))

class ObjectiveCompiler:
    def __init__(self, initial_dna, population_size=5):
        self.population = [ObjectiveDNA(initial_dna) for _ in range(population_size)]
        self.generation = 0
    
    def evaluate_fitness(self, feedback):
        for ind in self.population:
            ind.fitness_val = ind.fitness(feedback)
    
    def evolve(self, feedback, mutation_rate=0.2):
        self.evaluate_fitness(feedback)
        # Seleksi: pilih 2 terbaik
        self.population.sort(key=lambda x: x.fitness_val, reverse=True)
        parents = self.population[:2]
        # Rekombinasi + mutasi
        offspring = []
        for _ in range(len(self.population)-2):
            # simple crossover: pilih gen acak dari kedua parent
            genes1 = parents[0].genes.split(';')
            genes2 = parents[1].genes.split(';')
            new_genes = []
            for i in range(min(len(genes1), len(genes2))):
                if random.random() < 0.5:
                    new_genes.append(genes1[i])
                else:
                    new_genes.append(genes2[i])
            child = ObjectiveDNA(';'.join(new_genes))
            child = child.mutate(mutation_rate)
            offspring.append(child)
        self.population = parents + offspring
        self.generation += 1
        return self.population[0].genes  # return DNA terbaik
    
    def compile(self, dna):
        # Ubah DNA menjadi parameter sistem nyata
        params = {}
        for clause in dna.split(';'):
            if '>' in clause:
                key, val = clause.split('>')
                params[key] = ('min', float(val))
            elif '<' in clause:
                key, val = clause.split('<')
                params[key] = ('max', float(val))
        return params
```
