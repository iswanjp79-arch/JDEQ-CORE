# Modul Evaluasi oleh Auditor Jarvis & Cici
import pandas as pd
import matplotlib.pyplot as plt

def evaluasi_kinerja(data_log):
    df = pd.DataFrame(data_log)
    rata_rata_cpu = df['penggunaan_cpu'].mean()
    rata_rata_keamanan = df['skor_keamanan'].mean()
    
    print("=== LAPORAN EVALUASI AUDITOR ===")
    print(f"Rata-rata Penggunaan CPU: {rata_rata_cpu:.2f}%")
    print(f"Rata-rata Skor Keamanan: {rata_rata_keamanan:.2f}/100")
    
    if rata_rata_cpu > 40:
        rekomendasi = "Optimalkan modul Blavetip JEQ atau tambahkan sumber daya"
    else:
        rekomendasi = "Kinerja sistem sangat baik – pertahankan!"
    
    print(f"Rekomendasi: {rekomendasi}")
    plt.plot(df['waktu'], df['penggunaan_cpu'], label='Penggunaan CPU')
    plt.title('Grafik Kinerja Sistem Selama 1 Minggu')
    plt.savefig("$HOME/jeq/laporan_evaluasi.png")
    return rekomendasi
