from google import genai
import os

def muat_kunci():
    path = os.path.expanduser("~/.daftar_kunci.txt")
    if os.path.exists(path):
        with open(path, "r") as f:
            return [line.strip() for line in f if line.strip() and "AIza" in line]
    return []

def core_activation():
    kunci_list = muat_kunci()
    print("\n" + "="*40)
    print("      J.A.R.V.I.S  MULTI-KEY VAULT      ")
    print(f"      Ditemukan {len(kunci_list)} Kunci Aktif       ")
    print("="*40)

    for i, key in enumerate(kunci_list):
        try:
            client = genai.Client(api_key=key)
            response = client.models.generate_content(
                model="gemini-2.0-flash",
                contents="System check. Are you active?"
            )
            print(f"[SUCCESS]: Kunci #{i+1} Berhasil!")
            print(f"[REPLY]  : {response.text}")
            return # Mandek yen wis nemu sing bener
        except Exception as e:
            if "429" in str(e):
                print(f"[LIMIT]: Kunci #{i+1} Kuota Habis. Mencoba kunci lain...")
            else:
                print(f"[ERROR]: Kunci #{i+1} Bermasalah.")
    
    print("\n[CRITICAL]: Sedaya kunci sampun telas kuotane!")

if __name__ == "__main__":
    core_activation()
EOF
