#!/usr/bin/env python3
"Co-pilot (Mesin-2) - Generator kode taktis dan validator ringkas.File ini DIWAJIBKAN oleh conductor_termux.py"
import json
import uuid
from datetime import datetime
def generate_tactical_bundle(spt, blueprint):
"Simulasi Co-pilot: Membuat bundel taktis minimal berdasar SPT."
bundel_id = str(uuid.uuid4())
summary = f"Action(s): {', '.join(a['action'] for a in spt.get('actions', []))}"
# Skrip klasik yang akan dikirim ke Travelmate
script_lines = [
#!/bin/sh,
echo '== [TRAVELMATE SANDBOX] Eksekusi 'Bypass' Berhasil ==' ,
f"echo 'Intent dari Arsitek: {spt.get('intent')}'",
echo 'Timestamp: $(date)',
echo 'Menjalankan di dalam Docker Sandbox:',
ls -l /,
echo 'Siapa saya: $(id)',
echo 'Network (seharusnya kosong):',
ip addr show
]
if "pertahanan" in spt.get("intent", "").lower():
script_lines.append("echo 'Status: [DEFENSE_MODE_ACTIVATED] (simulated)'")
bundle = {
id: bundel_id,
created_at: datetime.utcnow().isoformat() + "Z",
spt_ref: spt.get("id"),
summary: summary,
tactics: spt.get("actions"),
classic_script: "\\n".join(script_lines),
metadata: {
generated_by: "co-pilot-nano-simulation",
blueprint_id: blueprint.get("id")
}
}
return bundle
def validate_bundle(bundle):
"Validator ringkas: cek keberadaan fields penting."
reasons = []
if not bundle.get("tactics"):
reasons.append("empty tactics")
if not bundle.get("classic_script"):
reasons.append("missing classic_script")
ok = len(reasons) == 0
return ok, reasons

#!/usr/bin/env python3
"OAI QUANTUM - Kondektor Termux-light (Vivo Y28)Arsitektur: M2M (Jalan Bypass) - Sesuai Log 2 & 3Kondektor (Vivo Y28) -> SSH -> Eksekutor (Travelmate)"
import argparse
import json
import uuid
import tempfile
import subprocess
import os
from pathlib import Path
from copilot_agent import generate_tactical_bundle, validate_bundle
from phi3_local import phi3_execute
from executor import Executor
BLUEPRINT_PATH = Path(__file__).parent / "blueprint.json"
def run_command(cmd, timeout=120):
"Menjalankan perintah subprocess lokal (seperti ssh/scp) di Termux."
try:
proc = subprocess.run(cmd, capture_output=True, text=True, timeout=timeout, check=True)
return {"returncode": proc.returncode, "stdout": proc.stdout, "stderr": proc.stderr}
except subprocess.CalledProcessError as e:
return {"returncode": e.returncode, "stdout": e.stdout, "stderr": e.stderr, "error": f"Command failed: {e}"}
except FileNotFoundError as e:
return {"returncode": -1, "stdout": "", "stderr": f"Perintah tidak ditemukan: {e.filename}. Pastikan openssh terinstal (pkg install openssh).", "error": str(e)}
except Exception as e:
return {"returncode": -1, "stdout": "", "stderr": f"Error tidak diketahui saat menjalankan perintah: {e}", "error": str(e)}
class ConductorTermux:
def __init__(self, blueprint_path=BLUEPRINT_PATH, simulate=True):
try:
with open(blueprint_path, "r", encoding="utf-8") as f:
self.blueprint = json.load(f)
except FileNotFoundError:
print(f"[FATAL] GAGAL MEMUAT PARTITUR: {blueprint_path} tidak ditemukan.")
self.blueprint = None # Akan menyebabkan kegagalan
self.simulate = simulate
self.executor = Executor(simulate=simulate) # Executor untuk mode --simulate
def intent_to_spt(self, intent_text):
"Layer 2: Menerjemahkan Visi Kuantum menjadi Vektor Perintah Taktis (SPT)."
spt = {
id: str(uuid.uuid4()),
intent: intent_text,
priority: "medium",
actions: []
}
if "pertahanan" in intent_text.lower():
spt["actions"].append({"action": "activate_defense_mode", "params": {"level": "standard"}})
else:
spt["actions"].append({"action": "log_intent", "params": {"raw": intent_text}})
return spt
def send_to_remote_and_execute(self, script_text, remote_user, remote_host,
remote_runner_path, ssh_key, timeout=300):
"Layer 4 (Bypass): Mengirim skrip ke Travelmate dan mengeksekusinya di Sandbox."
print(f"[JARVIS-M2M] Membuat file skrip sementara di Vivo Y28...")
# Membuat file skrip sementara di penyimpanan internal Termux
try:
tf = tempfile.NamedTemporaryFile("w", delete=False, prefix="oq_script_", suffix=".sh", dir=os.environ.get('TMPDIR', '/data/data/com.termux/files/usr/tmp'))
tf.write(script_text)
tf.flush()
os.fchmod(tf.fileno(), 0o700) # Pastikan file bisa dieksekusi
tf.close()
local_path = os.path.realpath(tf.name)
remote_tmp = f"/tmp/{Path(local_path).name}" # Path skrip di Travelmate
print(f"[JARVIS-M2M] File lokal dibuat: {local_path}")
except Exception as e:
return {"mode": "remote", "error": f"Gagal membuat file skrip sementara di Termux: {e}"}
try:
# === LANGKAH A: SCP (Kirim Skrip ke Travelmate) ===
print(f"[JARVIS-M2M] Mengirim {Path(local_path).name} ke {remote_user}@{remote_host}...")
scp_cmd = ["scp"]
if ssh_key:
scp_cmd += ["-i", ssh_key]
scp_cmd += [local_path, f"{remote_user}@{remote_host}:{remote_tmp}"]
scp_result = run_command(scp_cmd, timeout=60)
if scp_result["returncode"] != 0:
return {"mode": "remote", "error": "Gagal SCP (kirim skrip)", "detail": scp_result}
print("[JARVIS-M2M] Pengiriman skrip berhasil.")
# === LANGKAH B: SSH (Eksekusi Skrip di Travelmate) ===
print(f"[JARVIS-M2M] Mengeksekusi 'Kabel Bypass' di Travelmate...")
remote_command = f"bash -lc '{remote_runner_path} {remote_tmp}'"
ssh_cmd = ["ssh"]
if ssh_key:
ssh_cmd += ["-i", ssh_key]
ssh_cmd += [f"{remote_user}@{remote_host}", remote_command]
exec_result = run_command(ssh_cmd, timeout=timeout)
print("[JARVIS-M2M] Eksekusi remote selesai.")
# === LANGKAH C: SSH (Hapus Skrip Sementara di Travelmate) ===
print(f"[JARVIS-M2M] Membersihkan file sementara di Travelmate...")
cleanup_cmd = ["ssh"]
if ssh_key:
cleanup_cmd += ["-i", ssh_key]
cleanup_cmd += [f"{remote_user}@{remote_host}", f"rm -f {remote_tmp}"]
_ = run_command(cleanup_cmd, timeout=30) # Tidak perlu cek hasil, 'best-effort'
return {
mode: "remote_M2M_bypass",
scp_result: scp_result,
execution_result: exec_result,
remote_host: f"{remote_user}@{remote_host}"
}
finally:
# Selalu hapus file skrip sementara di Vivo Y28
try:
os.remove(local_path)
print(f"[JARVIS-M2M] Membersihkan file sementara di Vivo Y28.")
except Exception as e:
print(f"[PERINGATAN] Gagal menghapus file sementara lokal: {e}")
def orchestrate(self, intent_text, use_remote=False, remote_user=None, remote_host=None,
remote_runner_path="~/oq/run_sandbox.sh", ssh_key=None):
"Fungsi Orkestrasi Utama (Kondektor)."
if not self.blueprint:
return {"status": "fatal_error", "error": "Blueprint (Partitur) tidak ter-load. Tidak bisa melanjutkan."}
print(f"--- [OAI QUANTUM KONDEKTOR AKTIF] ---")
print(f"[JARVIS] Menerima Visi Kuantum: '{intent_text}'")
# Layer 2: Menerjemahkan Intent
spt = self.intent_to_spt(intent_text)
print(f"[JARVIS] Vektor SPT (L2) dibuat: {spt['id']}")
# Layer 3: Simulasi Agen Taktis (Co-pilot & Phi-3)
tactical_bundle = generate_tactical_bundle(spt, self.blueprint)
print(f"[JARVIS] Bundel Taktis (L3) diterima: {tactical_bundle['summary']}")
valid, reasons = validate_bundle(tactical_bundle)
if not valid:
print(f"[JARVIS-GAGAL] Validasi bundel gagal: {reasons}")
return {"status": "validation_failed", "reasons": reasons}
phi3_response = phi3_execute(tactical_bundle)
print(f"[JARVIS] Verifikasi Phi-3 (L3) (Lokal): {phi3_response['verdict']}")
# Layer 4: Logika Eksekusi (Bypass atau Simulasi)
if use_remote:
if not (remote_user and remote_host):
print(f"[JARVIS-GAGAL] Mode 'use-remote' aktif tapi --remote-user atau --remote-host tidak disetel.")
return {"status": "error", "error": "remote_user_and_host_required"}
print(f"[JARVIS] MEMULAI 'BYPASS' (L4) ke Travelmate...")
exec_result = self.send_to_remote_and_execute(
tactical_bundle.get("classic_script", "# no-op\\n"),
remote_user=remote_user,
remote_host=remote_host,
remote_runner_path=remote_runner_path,
ssh_key=ssh_key
)
else:
print(f"[JARVIS] Menjalankan Mode SIMULASI LOKAL (L4)...")
exec_result = self.executor.run(tactical_bundle.get("classic_script", "# no-op\\n"))
# Layer 5: Umpan Balik
feedback = {
spt_id: spt["id"],
tactical_id: tactical_bundle.get("id"),
phi3_verification: phi3_response,
execution_result: exec_result
}
print(f"[JARVIS] Orkestrasi Selesai (L5). Umpan balik dibuat.")
print(f"--- [KONDEKTOR SELESAI] ---")
return {"status": "ok", "feedback": feedback}
def main():
p = argparse.ArgumentParser(description="OAI QUANTUM Kondektor Termux (M2M Bypass)")
p.add_argument("--intent", type=str, default="Aktifkan pertahanan", help="Visi Kuantum dari Arsitek")
p.add_argument("--simulate", type=lambda s: s.lower() in ("1","true","yes"), default=True, help="Mode simulasi lokal (default: True)")

p.add_argument("--remote-user", type=str, help="Username SSH di Travelmate")

p.add_argument("--remote-runner-path", type=str, default="~/oq/run_sandbox.sh", help="Path ke skrip 'run_sandbox.sh' di Travelmate")
p.add_argument("--ssh-key", type=str, default="~/.ssh/id_oq", help="Path ke kunci private SSH (e.g., ~/.ssh/id_oq)")
args = p.parse_args()
# Logika 'Zero Failure' untuk mode
# Jika --use-remote TIDAK disetel, --simulate akan otomatis True (default).
# Jika --use-remote DISETEL, kita setel simulate=False agar Kondektor tahu mode 'real' aktif.
is_simulating = args.simulate
if args.use_remote:
is_simulating = False # Mode 'use-remote' adalah mode 'real'

remote_user=args.remote_user, remote_host=args.remote_host,
remote_runner_path=args.remote_runner_path, ssh_key=args.ssh_key)
print("\\n=== [HASIL FINAL ORKESTRASI (JSON)] ===")
import pprint; pprint.pprint(result)
print("========================================")
if __name__ == "__main__":
main()
